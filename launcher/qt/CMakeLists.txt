cmake_minimum_required(VERSION 3.21)
project(KirbyQtLauncher LANGUAGES CXX)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
qt_add_executable(kirby-ui WIN32 main.cpp window.cpp controls.cpp theme.cpp)
target_link_libraries(kirby-ui PRIVATE Qt6::Widgets)
target_compile_features(kirby-ui PRIVATE cxx_std_20)
set_target_properties(kirby-ui PROPERTIES AUTORCC ON)
target_sources(kirby-ui PRIVATE artwork.qrc)
if(WIN32)
    enable_language(RC)
    target_sources(kirby-ui PRIVATE ../launcher.rc)
    target_include_directories(kirby-ui PRIVATE ..)
    # Keep this kit's runtime separate from the game's LLVM-MinGW libraries.
    get_target_property(qmake Qt6::qmake IMPORTED_LOCATION)
    get_filename_component(qt_bin "${qmake}" DIRECTORY)
    get_filename_component(compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set(notices "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/licenses")
    if(NOT CMAKE_RUNTIME_OUTPUT_DIRECTORY)
        set(notices "${CMAKE_CURRENT_BINARY_DIR}/licenses")
    endif()
    file(MAKE_DIRECTORY "${notices}")
    configure_file("../../LICENSE" "${notices}/LICENSE-project.txt" COPYONLY)
    configure_file("../../licenses/LICENSE-launcher-font.txt" "${notices}/LICENSE-launcher-font.txt" COPYONLY)
    foreach(module qtbase qtsvg qttranslations)
        set(record "${qt_bin}/../sbom/${module}-${Qt6_VERSION}.spdx")
        if(EXISTS "${record}")
            file(COPY "${record}" DESTINATION "${notices}")
        endif()
    endforeach()
    foreach(component gcc mingw-w64 winpthreads)
        if(IS_DIRECTORY "${compiler_bin}/../licenses/${component}")
            file(COPY "${compiler_bin}/../licenses/${component}" DESTINATION "${notices}/compiler")
        endif()
    endforeach()
    add_custom_command(TARGET kirby-ui POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E env "PATH=${compiler_bin};${qt_bin};$ENV{PATH}"
        "${qt_bin}/windeployqt.exe" --release --no-opengl-sw --no-system-d3d-compiler
        --no-quick-import --include-plugins qoffscreen --translations zh_CN,es "$<TARGET_FILE:kirby-ui>" VERBATIM)
endif()
